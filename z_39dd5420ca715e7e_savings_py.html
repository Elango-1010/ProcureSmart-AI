12def contract_renewal_alerts(engine=None) -> pd.DataFrame: 

13 """Contracts expiring within the renewal warning window.""" 

14 contracts = read_table("contracts", engine) 

15 contracts["end_date"] = pd.to_datetime(contracts["end_date"]) 

16 today = pd.Timestamp(datetime.utcnow().date()) 

17 

18 contracts["days_to_expiry"] = (contracts["end_date"] - today).dt.days 

19 due = contracts[ 

20 (contracts["days_to_expiry"] <= CONTRACT_RENEWAL_WARNING_DAYS) 

21 & (contracts["status"] == "ACTIVE") 

22 ].copy() 

23 due["priority"] = due["days_to_expiry"].apply( 

24 lambda d: "HIGH" if d <= 30 else ("MEDIUM" if d <= 60 else "LOW") 

25 ) 

26 due["recommendation"] = due.apply( 

27 lambda r: ( 

28 f"Contract {r['contract_id']} expires in {int(r['days_to_expiry'])} days " 

29 f"(current discount {r['negotiated_discount_pct']}%). Begin renegotiation now to " 

30 "lock in or improve terms before lapse." 

31 ), 

32 axis=1, 

33 ) 

34 return due.sort_values("days_to_expiry").reset_index(drop=True)

37def volume_milestone_opportunities(engine=None) -> pd.DataFrame: 

38 """Suppliers approaching a volume milestone that should trigger renegotiation.""" 

39 spend = read_table("spend_transactions", engine) 

40 by_supplier = ( 

41 spend.groupby(["supplier_id", "supplier_name"], as_index=False)["amount"].sum() 

42 .rename(columns={"amount": "total_spend"}) 

43 ) 

44 

45 rows = [] 

46 for _, row in by_supplier.iterrows(): 

47 spend_amt = row["total_spend"] 

48 passed = [m for m in VOLUME_MILESTONES if spend_amt >= m] 

49 next_milestones = [m for m in VOLUME_MILESTONES if spend_amt < m] 

50 if not passed: 

51 continue 

52 current_milestone = passed[-1] 

53 next_milestone = next_milestones[0] if next_milestones else None 

54 gap_to_next = round(next_milestone - spend_amt, 2) if next_milestone else None 

55 

56 rows.append({ 

57 "supplier_id": row["supplier_id"], 

58 "supplier_name": row["supplier_name"], 

59 "total_spend": round(float(spend_amt), 2), 

60 "milestone_reached": current_milestone, 

61 "next_milestone": next_milestone, 

62 "gap_to_next_milestone": gap_to_next, 

63 "recommendation": ( 

64 f"{row['supplier_name']} has crossed the Rs {current_milestone:,.0f} spend milestone " 

65 f"(Rs {spend_amt:,.0f} YTD). This volume typically qualifies for improved pricing - " 

66 "trigger a renegotiation review." 

67 ), 

68 }) 

69 

70 return pd.DataFrame(rows).sort_values("total_spend", ascending=False).reset_index(drop=True)
